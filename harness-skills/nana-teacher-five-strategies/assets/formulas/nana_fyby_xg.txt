STF:=NOT(NAMELIKE('ST') OR NAMELIKE('*ST') OR NAMELIKE('退'));
LISTED:=BARSCOUNT(C)>120;
LIQ:=AMOUNT>=30000000;
AVGPRICE:=IF(V>0,AMOUNT/V/100,C);
VISIBLE:=O>REF(C,1)*1.01 AND C<O AND C>REF(C,1) AND AVGPRICE>REF(C,1);
HIDDEN:=O>REF(C,1)*1.01 AND C<O AND C<=REF(C,1) AND AVGPRICE>REF(C,1);
FAKE:=VISIBLE OR HIDDEN;
SAFEVOL:=V>REF(V,1)*1.05 AND V<MA(V,20)*3;
FAKEOK:=FAKE AND SAFEVOL AND (C>MA(C,5) OR MA(C,5)>MA(C,10));

TREND:=MA(C,5)>MA(C,10) AND MA(C,10)>=MA(C,20);
DIF:=EMA(C,12)-EMA(C,26);
DEA:=EMA(DIF,9);
MACDOK:=CROSS(DIF,DEA) OR DIF>DEA;
RSV:=(C-LLV(L,9))/(HHV(H,9)-LLV(L,9))*100;
K:=SMA(RSV,3,1);
D:=SMA(K,3,1);
KDJOK:=CROSS(K,D) OR K>D;
CONFIRM:=REF(FAKEOK,1) AND C>O AND V<REF(V,1)*0.70 AND C>REF(C,2) AND L>REF(L,1)*0.97;
RECLAIM:=REF(FAKEOK,1) AND C>REF(O,1) AND C>O AND V>REF(V,1)*0.80 AND L>REF(L,1)*0.97;

XG:(CONFIRM OR RECLAIM) AND TREND AND (MACDOK OR KDJOK) AND STF AND LISTED AND LIQ AND C>3;
