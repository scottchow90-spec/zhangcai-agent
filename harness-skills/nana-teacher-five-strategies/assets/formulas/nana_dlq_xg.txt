STF:=NOT(NAMELIKE('ST') OR NAMELIKE('*ST') OR NAMELIKE('退'));
LISTED:=BARSCOUNT(C)>120;
LIQ:=AMOUNT>=30000000;
VOLLOW:=V<MA(V,20)*0.60;
NARROW:=HHV(H,5)/LLV(L,5)<1.08;
GROUP:=COUNT(VOLLOW,8)>=5 AND COUNT(NARROW,8)>=4;
NOLOW:=L>REF(LLV(L,60),1)*0.98;
MAUP:=MA(C,10)>=REF(MA(C,10),3) AND C>MA(C,5);

DIF:=EMA(C,12)-EMA(C,26);
DEA:=EMA(DIF,9);
MACDOK:=CROSS(DIF,DEA) OR DIF>DEA;
RSV:=(C-LLV(L,9))/(HHV(H,9)-LLV(L,9))*100;
K:=SMA(RSV,3,1);
D:=SMA(K,3,1);
KDJOK:=CROSS(K,D) OR K>D;
BOLL:=MA(C,20);
BOLLUP:=C>BOLL;
CONFIRM:=MACDOK+KDJOK+BOLLUP>=2;
BREAKOUT:=C>O AND C/REF(C,1)>1.03 AND V>MA(V,5)*1.20 AND C>MA(C,20);

XG:GROUP AND NOLOW AND MAUP AND (BREAKOUT OR (CROSS(C,MA(C,5)) AND CONFIRM)) AND STF AND LISTED AND LIQ AND C>2;
