STF:=NOT(NAMELIKE('ST') OR NAMELIKE('*ST') OR NAMELIKE('退'));
LISTED:=BARSCOUNT(C)>120;
LIQ:=AMOUNT>=30000000;
TREND:=MA(C,5)>MA(C,10) AND MA(C,10)>=MA(C,20);
LIMITUP:=C/REF(C,1)>=1.095 AND C=H;

YBPOS3:=REF(LIMITUP,3) AND REF(LLV(L,2),1)>=REF((O+C)/2,3) AND C>REF(H,1) AND C>O AND V>MA(V,5)*1.20;
YBPOS4:=REF(LIMITUP,4) AND REF(LLV(L,3),1)>=REF((O+C)/2,4) AND C>REF(H,1) AND C>O AND V>MA(V,5)*1.20;
YBPOS5:=REF(LIMITUP,5) AND REF(LLV(L,4),1)>=REF((O+C)/2,5) AND C>REF(H,1) AND C>O AND V>MA(V,5)*1.20;
YBPOS6:=REF(LIMITUP,6) AND REF(LLV(L,5),1)>=REF((O+C)/2,6) AND C>REF(H,1) AND C>O AND V>MA(V,5)*1.20;
YBPOS7:=REF(LIMITUP,7) AND REF(LLV(L,6),1)>=REF((O+C)/2,7) AND C>REF(H,1) AND C>O AND V>MA(V,5)*1.20;
YBPOS:=YBPOS3 OR YBPOS4 OR YBPOS5 OR YBPOS6 OR YBPOS7;

YBNEG3:=REF(C,3)<REF(O,3) AND (REF(O,3)-REF(C,3))/REF(O,3)>0.05 AND REF(V,3)>REF(MA(V,10),3)*1.20 AND REF(HHV(MAX(O,C),2),1)<=REF(O,3) AND REF(LLV(MIN(O,C),2),1)>=REF(C,3) AND C>REF(O,3) AND C>O AND V>MA(V,5)*1.20;
YBNEG4:=REF(C,4)<REF(O,4) AND (REF(O,4)-REF(C,4))/REF(O,4)>0.05 AND REF(V,4)>REF(MA(V,10),4)*1.20 AND REF(HHV(MAX(O,C),3),1)<=REF(O,4) AND REF(LLV(MIN(O,C),3),1)>=REF(C,4) AND C>REF(O,4) AND C>O AND V>MA(V,5)*1.20;
YBNEG5:=REF(C,5)<REF(O,5) AND (REF(O,5)-REF(C,5))/REF(O,5)>0.05 AND REF(V,5)>REF(MA(V,10),5)*1.20 AND REF(HHV(MAX(O,C),4),1)<=REF(O,5) AND REF(LLV(MIN(O,C),4),1)>=REF(C,5) AND C>REF(O,5) AND C>O AND V>MA(V,5)*1.20;
YBNEG:=YBNEG3 OR YBNEG4 OR YBNEG5;

XG:(YBPOS OR YBNEG) AND STF AND LISTED AND LIQ AND TREND AND C>3;
