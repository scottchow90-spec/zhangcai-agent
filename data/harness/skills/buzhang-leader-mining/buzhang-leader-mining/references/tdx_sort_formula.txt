LIM0:=IF(CODELIKE('688') OR CODELIKE('300'),0.2,IF(NAMEINCLUDE('ST') OR NAMEINCLUDE('*ST'),0.05,0.1));
ZT0:=C>=ZTPRICE(REF(C,1),LIM0) AND C=H;
BR0:=H>=ZTPRICE(REF(C,1),LIM0) AND C<ZTPRICE(REF(C,1),LIM0);
BC0:=IF(ZT0,1+BARSLAST(NOT(ZT0)),0);
HS0:=INBLOCK('热点主线龙头挖掘');
HD1:=INBLOCK('热点方向第1名');
HD2:=INBLOCK('热点方向第2名');
HD3:=INBLOCK('热点方向第3名');
RANK0:=IF(HD1,1,IF(HD2,2,IF(HD3,3,0)));
RL1:=INBLOCK('补涨龙1');
RL2:=INBLOCK('补涨龙2');
RL3:=INBLOCK('补涨龙3');
ROLE0:=RL1 OR RL2 OR RL3;
DIR0:=HD1 OR HD2 OR HD3;
RISK0:=NOT(NAMELIKE('ST') OR NAMEINCLUDE('*ST') OR NAMEINCLUDE('退') OR NAMEINCLUDE('退市'));
EXCL0:=IF(RISK0=0,1,0);
BARS0:=BARSCOUNT(C);
RET1:=IF(BARS0>1,C/REF(C,1)-1,0);
RET5:=IF(BARS0>5,C/REF(C,5)-1,0);
AMTR0:=IF(AMOUNT>0,AMOUNT/MAX(EMA(AMOUNT,20),1),0);
CLV0:=IF(H>L,(2*C-H-L)/(H-L),0);
MOMR0:=RET5;
MAPR0:=RET1;
LADR0:=BC0+CLV0;
CAPR0:=AMTR0*CLV0;
HH20:=HHV(C,20);
LL20:=LLV(C,20);
POSR0:=IF(HH20>LL20,(HH20-C)/(HH20-LL20),0);
VRNG0:=HHV(RET1,20)-LLV(RET1,20);
RISKR0:=IF(BR0,0,1/(1+ABS(RET5)+VRNG0));
MOMN0:=IF(HHV(MOMR0,60)>LLV(MOMR0,60),(MOMR0-LLV(MOMR0,60))/(HHV(MOMR0,60)-LLV(MOMR0,60))*100,0);
MAPN0:=IF(HHV(MAPR0,60)>LLV(MAPR0,60),(MAPR0-LLV(MAPR0,60))/(HHV(MAPR0,60)-LLV(MAPR0,60))*100,0);
LADN0:=IF(HHV(LADR0,60)>LLV(LADR0,60),(LADR0-LLV(LADR0,60))/(HHV(LADR0,60)-LLV(LADR0,60))*100,0);
CAPN0:=IF(HHV(CAPR0,60)>LLV(CAPR0,60),(CAPR0-LLV(CAPR0,60))/(HHV(CAPR0,60)-LLV(CAPR0,60))*100,0);
POSN0:=POSR0*100;
RISKN0:=IF(HHV(RISKR0,60)>LLV(RISKR0,60),(RISKR0-LLV(RISKR0,60))/(HHV(RISKR0,60)-LLV(RISKR0,60))*100,0);
STOCK0:=IF(BARS0>1,IF(C>REF(C,1) AND AMTR0>REF(AMTR0,1),1,0),0);
QUAL0:=IF(HS0 AND DIR0 AND STOCK0 AND EXCL0=0,1,0);
CORE0:=(MOMN0+MAPN0+LADN0+CAPN0+POSN0)/5;
SORT0:=IF(QUAL0=1,CORE0*RISKN0/100,0);
热点闸门:QUAL0,NODRAW;
热点强度:MOMN0,NODRAW;
直接映射:MAPN0,NODRAW;
梯队质量:LADN0,NODRAW;
资金承接:CAPN0,NODRAW;
相对位置:POSN0,NODRAW;
风险执行:RISKN0,NODRAW;
剔除标记:EXCL0,NODRAW;
补涨总分:CORE0,NODRAW,COLORRED;
龙位编码:IF(QUAL0=1,IF(RL1,1,IF(RL2,2,IF(RL3,3,0))),0),NODRAW;
龙1候选:IF(QUAL0=1 AND RL1,SORT0,0),NODRAW,COLORRED;
龙2候选:IF(QUAL0=1 AND RL2,SORT0,0),NODRAW,COLORYELLOW;
龙3候选:IF(QUAL0=1 AND RL3,SORT0,0),NODRAW,COLORGREEN;
热点板块码:IF(QUAL0=1,RANK0,0),NODRAW;
排序键:SORT0,NODRAW,COLORWHITE;
